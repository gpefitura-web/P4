10:008  def synthetic_solar(seed=42):
10:009      """
10:010      Synthetic daily solar S-index, 1966-2021, calibrated on:
10:011        Egeland+2017  -- mean level S = 0.1694, residual scatter ~0.003
10:012        Usoskin+2017  -- 11 yr period; cycle 24 the weakest in ~100 yr
10:013  
10:014      Construction:  S(t) = S_mean + A(t) sin(omega t) + eps(t; phi_eff(t))
10:015  
10:016      The noise term carries a time-varying AR(1) persistence phi_eff(t) that rises
10:017      from 0.35 to 0.72 around 2003-2010. This embeds the CSD signature: near a
10:018      bifurcation the recovery time tau = -1/lambda diverges, which appears as
10:019      growing lag-1 autocorrelation in the stochastic component above the cycle.
10:020      """
10:021      rng = np.random.default_rng(seed)
10:022      N = 20440                                  # 1966-01-01 .. 2021-12-31
10:023      t = np.arange(N, dtype=float)
10:024      yrs = 1966.0 + t / 365.25
10:025  
10:026      # Per-cycle amplitude envelope
10:027      amps = [(1966, 1976, 0.025), (1976, 1986, 0.025), (1986, 1996, 0.024),
10:028              (1996, 2008, 0.019), (2008, 2019, 0.009), (2019, 2030, 0.017)]
10:029      A = np.zeros(N)
10:030      for y0, y1, a in amps:
10:031          A[(yrs >= y0) & (yrs < y1)] = a
10:032      A = uniform_filter1d(A, size=365 * 2)      # smooth cycle-to-cycle steps
10:033  
10:034      omega = 2 * np.pi / S_CYCLE_DAYS
10:035      phase = 2 * np.pi * (1968.0 - 1966.0) / 11.0
10:036      s_cycle = S_SUN_MEAN + A * np.sin(omega * t - phase)
10:037  
10:038      # CSD bump in AR(1) persistence, centred on 2007
10:039      t_csd = (2007.0 - 1966.0) * 365.25
10:040      phi_eff = 0.35 + 0.37 / (1 + np.exp((np.abs(t - t_csd) - 4 * 365.25) / 365.0))
10:041  
10:042      eps = np.zeros(N)
10:043      for i in range(1, N):
10:044          eps[i] = (phi_eff[i] * eps[i-1]
10:045                    + 0.003 * np.sqrt(max(0.0, 1 - phi_eff[i]**2)) * rng.standard_normal())
10:046  
10:047      return yrs, np.clip(s_cycle + eps, 0.12, 0.24)

12:001  def _seed(hd):
12:002      return int(hd) % (2**31 - 1)

12:004  def synthetic_star(hd, s_mean, s_std, p_cyc_yr, group, N=26):
12:005      """
12:006      Synthetic seasonal S-index series (1966-1991, ~1 point per year).
12:007  
12:008      The CSD hypothesis is encoded in the per-group AR(1) persistence:
12:009        Cycling   -- regular cycle plus weak AR(1) noise (phi ~ 0.25)
12:010        Irregular -- phi_1 in [0.50, 0.75]: long memory, approaching bifurcation
12:011        Flat      -- phi_1 in [0.02, 0.15]: new stable fixed point, fast relaxation
12:012      Rationale: Dakos+2012 Sec. 2 (rising AR1 as the transition is approached);
12:013      Scheffer+2009 Fig. 2.
12:014      """
12:015      rng = np.random.default_rng(_seed(hd))
12:016      yrs = np.arange(1966, 1966 + N, dtype=float)
12:017  
12:018      if group == "Cycling":
12:019          if p_cyc_yr > 0:
12:020              phase = rng.uniform(0, 2 * np.pi)
12:021              s = s_mean + s_std * np.sqrt(2) * np.sin(2 * np.pi / p_cyc_yr * yrs + phase)
12:022          else:
12:023              s = np.full(N, s_mean)
12:024          phi_c = 0.25
12:025          noise = np.zeros(N)
12:026          noise[0] = rng.standard_normal() * s_std * 0.2
12:027          for i in range(1, N):
12:028              noise[i] = (phi_c * noise[i-1]
12:029                          + s_std * 0.2 * np.sqrt(1 - phi_c**2) * rng.standard_normal())
12:030          s = s + noise
12:031  
12:032      elif group == "Irregular":
12:033          phi = rng.uniform(0.50, 0.75)
12:034          s = np.zeros(N); s[0] = s_mean
12:035          scale = s_std * 1.2
12:036          for i in range(1, N):
12:037              s[i] = (s_mean + phi * (s[i-1] - s_mean)
12:038                      + scale * np.sqrt(1 - phi**2) * rng.standard_normal())
12:039  
12:040      else:  # Flat
12:041          phi = rng.uniform(0.02, 0.15)
12:042          s = np.zeros(N); s[0] = s_mean
12:043          scale = s_std * 0.8
12:044          for i in range(1, N):
12:045              s[i] = (s_mean + phi * (s[i-1] - s_mean)
12:046                      + scale * np.sqrt(1 - phi**2) * rng.standard_normal())
12:047  
12:048      return yrs, np.clip(s, 0.05, 0.80)

13:001  ## 5 · Analysis functions
13:002  
13:003  ### Detrending policy — two distinct tracks
13:004  
13:005  Dakos et al. (2012), Table 3 footnote 5, is explicit that DFA takes **polynomial
13:006  detrending within the rolling window only**. Gaussian pre-filtering is appropriate for
13:007  the AR(1) and variance indicators, *not* for DFA. The two tracks therefore differ:
13:008  
13:009  | Track | Indicator | Detrending |
13:010  |---|---|---|
13:011  | **A** | DFA → H | linear detrend of the full series; polynomial detrend inside each DFA box |
13:012  | **B** | AR(1), σ² | Gaussian kernel (σ = 1 yr) removes the cycle *first*, then linear detrend inside each rolling window |
13:013  
13:014  DFA is implemented natively following Peng et al. (1994) — no external dependency.

14:001  def dfa(x, nvals=None, order=1, return_fit=False):
14:002      """
14:003      Detrended Fluctuation Analysis -- Peng et al. 1994, Phys. Rev. E 49, 1685.
14:004  
14:005      Algorithm (Peng+1994 Sec. II):
14:006        1. integrate the mean-centred series into the profile Y(i)
14:007        2. split Y into non-overlapping boxes of length n
14:008        3. fit and subtract a polynomial of `order` WITHIN each box
14:009        4. F(n) = RMS of the detrended profile
14:010        5. alpha (= Hurst exponent H) is the slope of log F(n) vs log n
14:011  
14:012      Scale range: n_min = 4, because below n = 4 the detrending algorithm itself
14:013      distorts the power law (Peng+1994, p. 1688). n_max = N/4 so that every scale
14:014      is averaged over at least four boxes.
14:015  
14:016      Returns alpha, or (alpha, n_values, F_values) when return_fit=True.
14:017      """
14:018      x = np.asarray(x, dtype=float)
14:019      x = x[np.isfinite(x)]
14:020      N = len(x)
14:021      if N < 20:
14:022          return (np.nan, None, None) if return_fit else np.nan
14:023  
14:024      Y = np.cumsum(x - np.mean(x))                      # step 1
14:025  
14:026      if nvals is None:                                  # step 2
14:027          n_min, n_max = 4, max(8, N // 4)
14:028          nvals = np.unique(np.floor(
14:029              np.logspace(np.log10(n_min), np.log10(n_max), 20)).astype(int))
14:030          nvals = nvals[nvals >= order + 2]
14:031  
14:032      F, used = [], []
14:033      for n in nvals:
14:034          n_boxes = N // n
14:035          if n_boxes < 4:
14:036              continue
14:037          idx = np.arange(n, dtype=float)
14:038          resid_sq = []
14:039          for b in range(n_boxes):                       # steps 3 and 4
14:040              seg = Y[b*n:(b+1)*n]
14:041              trend = np.polyval(np.polyfit(idx, seg, order), idx)
14:042              resid_sq.append(np.mean((seg - trend)**2))
14:043          F.append(np.sqrt(np.mean(resid_sq)))
14:044          used.append(n)
14:045  
14:046      if len(used) < 4:
14:047          return (np.nan, None, None) if return_fit else np.nan
14:048  
14:049      alpha, _ = np.polyfit(np.log10(used), np.log10(F), 1)   # step 5
14:050      if return_fit:
14:051          return float(alpha), np.array(used), np.array(F)
14:052      return float(alpha)
14:053  
14:054  def dfa_hurst(series):
14:055      """
14:056      DFA wrapper returning (H, note).
14:057  
14:058      A series of length N only admits scales 4 <= n <= N/4, and at least four
14:059      distinct scales are needed for a meaningful log-log slope. That requires
14:060      N >~ 100 -- exactly the threshold Dakos+2012 state for robust DFA. Shorter
14:061      series return NaN rather than a number that cannot be defended.
14:062      """
14:063      n = len(series)
14:064      if n < 100:
14:065          return np.nan, f"N={n} < 100: DFA not defined (Dakos+2012)"
14:066      return dfa(series), ""
14:067  
14:068  def ar1_coef(x):
14:069      """Lag-1 autocorrelation (equivalently the AR(1) coefficient phi_1)."""
14:070      if len(x) < 4:
14:071          return np.nan
14:072      return float(np.corrcoef(x[:-1], x[1:])[0, 1])
14:073  
14:074  def sliding_ews(series, window_frac=0.50, step=1):
14:075      """
14:076      Rolling-window AR(1) and normalised variance (Dakos+2012 Sec. 2).
14:077      Applies a linear detrend inside each window. Feed it the CYCLE-REMOVED residuals.
14:078      Returns (centre_indices, phi1, var_norm).
14:079      """
14:080      N = len(series)
14:081      w = max(4, int(round(N * window_frac)))
14:082      centres, phi1_arr, var_arr = [], [], []
14:083      for start in range(0, N - w + 1, step):
14:084          chunk = signal.detrend(series[start:start + w], type="linear")
14:085          centres.append((2 * start + w) // 2)
14:086          phi1_arr.append(ar1_coef(chunk))
14:087          var_arr.append(float(np.var(chunk, ddof=1)))
14:088      var_arr = np.array(var_arr)
14:089      return (np.array(centres), np.array(phi1_arr),
14:090              var_arr / (np.nanmean(var_arr) + 1e-12))
14:091  
14:092  def kendall_tau(metric):
14:093      """Kendall tau trend test. tau > 0 means the indicator is rising (CSD signal)."""
14:094      x = np.arange(len(metric))
14:095      ok = np.isfinite(metric)
14:096      if ok.sum() < 4:
14:097          return np.nan, np.nan
14:098      tau, p = stats.kendalltau(x[ok], metric[ok])
14:099      return float(tau), float(p)
14:100  
14:101  def remove_cycle(s, sigma_yr=1.0, dt_days=1.0):
14:102      """
14:103      Remove the deterministic activity cycle with a Gaussian kernel smoother
14:104      (Dakos+2012 Sec. 2 / Fig. 2). Returns (residuals, trend).
14:105      Required before AR(1) and variance -- NOT applied before DFA.
14:106      """
14:107      trend = gaussian_filter1d(s, sigma=sigma_yr * 365.25 / dt_days, mode="reflect")
14:108      return s - trend, trend
14:109  
14:110  def spectral_beta(s, fs=1.0):
14:111      """
14:112      Power-spectral exponent beta from S(f) ~ f^-beta, fitted between 1/(5 yr)
14:113      and 1/(0.5 yr) -- above the cycle peak, below the white-noise floor.
14:114      Related to the Hurst exponent by H = (beta + 1)/2.
14:115      """
14:116      from scipy.signal import welch
14:117      f, psd = welch(s, fs=fs, nperseg=min(4096, max(256, len(s) // 4)))
14:118      mask = (f > 1 / (5 * 365.25)) & (f < 1 / (0.5 * 365.25)) & (psd > 0)
14:119      if mask.sum() < 5:
14:120          return np.nan, np.nan, np.nan
14:121      slope, _, r, _, se = stats.linregress(np.log10(f[mask]), np.log10(psd[mask]))
14:122      return float(-slope), float(se), float(r**2)
14:123

20:001  def mw_test(df, g1, g2, metric, alt="greater"):
20:002      a = df[df.Group == g1][metric].dropna().values
20:003      b = df[df.Group == g2][metric].dropna().values
20:004      if len(a) < 2 or len(b) < 2:
20:005          return dict(g1=g1, g2=g2, metric=metric, U=np.nan, p=np.nan,
20:006                      med1=np.nan, med2=np.nan, sig="--", n1=len(a), n2=len(b))

20:008      sig = "***" if p < 0.001 else "**" if p < 0.01 else "*" if p < 0.05 else "n.s."
20:009      return dict(g1=g1, g2=g2, metric=metric, U=float(U), p=float(p),
20:010                  med1=float(np.median(a)), med2=float(np.median(b)),
20:011                  sig=sig, n1=len(a), n2=len(b))
